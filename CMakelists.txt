cmake_minimum_required(VERSION 3.10)
project(SM_Shared_Project C)

# 1. 동적 라이브러리 생성 (타깃명: sm)
add_library(sm SHARED sm.c sm.h)

# [핵심] 모든 OS에서 라이브러리 이름을 'sm'으로 강제 통일
# - PREFIX "" 설정을 통해 리눅스/macOS에서 'libsm.so'가 아닌 'sm.so'로 생성됩니다.
set_target_properties(sm PROPERTIES 
    OUTPUT_NAME "sm"
    PREFIX ""
)

# 2. 실행 파일 생성 (타깃명: sm)
add_executable(sm main.c)

# [핵심] 모든 OS에서 실행 파일 이름을 'sm'으로 강제 통일
set_target_properties(sm PROPERTIES 
    OUTPUT_NAME "sm"
)

# 3. 실행 파일에 동적 라이브러리 연결
target_link_libraries(sm PRIVATE sm)
