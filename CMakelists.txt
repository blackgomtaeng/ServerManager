cmake_minimum_required(VERSION 3.10)
project(SM_Shared_Project C)

# 1. 동적 라이브러리 생성 (타깃명: sm_lib)
add_library(sm_lib SHARED sm.c sm.h)

# 리눅스/macOS에서도 libsm.so 대신 sm.so로 생성되도록 설정
set_target_properties(sm_lib PROPERTIES 
    OUTPUT_NAME "sm"
    PREFIX ""
)

# 윈도우 환경인 경우 소켓 라이브러리 링크 추가
if(WIN32)
    target_link_libraries(sm_lib PRIVATE ws2_32)
endif()

# 2. 실행 파일 생성 (타깃명: sm_exe)
add_executable(sm_exe main.c)

# 실행 파일의 결과물 이름을 'sm'(윈도우는 sm.exe)으로 강제 통일
set_target_properties(sm_exe PROPERTIES OUTPUT_NAME "sm")

# 3. 실행 파일에 동적 라이브러리 연결 (타깃명 sm_lib를 지정)
target_link_libraries(sm_exe PRIVATE sm_lib)
